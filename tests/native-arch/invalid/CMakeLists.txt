cmake_minimum_required(VERSION 3.22)
project(GhajarInvalidNativeArch LANGUAGES C)
include("${CMAKE_CURRENT_LIST_DIR}/../../../openvpn/src/main/cpp/ghajar-openssl-arch.cmake")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generic.c" "int generic(void) { return 0; }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ecp_sm2p256.c" "int optimized(void) { return 0; }\n")
add_library(crypto STATIC "${CMAKE_CURRENT_BINARY_DIR}/generic.c")
if(NOT CASE STREQUAL "missing_source")
    target_sources(crypto PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/ecp_sm2p256.c")
endif()
set(abi "armeabi-v7a")
if(CASE STREQUAL "unknown_abi")
    set(abi "unreviewed")
elseif(CASE STREQUAL "wrong_asm")
    target_compile_definitions(crypto PRIVATE ECP_SM2P256_ASM)
elseif(CASE STREQUAL "missing_asm")
    set(abi "arm64-v8a")
endif()
ghajar_harden_openssl(crypto "${abi}")
